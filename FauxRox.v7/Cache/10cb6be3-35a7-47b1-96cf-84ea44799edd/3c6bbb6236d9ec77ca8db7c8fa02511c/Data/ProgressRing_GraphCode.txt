// ============================================================================
// Progress Ring - Graph Material Custom Code Node
// ============================================================================
// Lens Studio'da Material oluşturduktan sonra:
// 1. Material seç → Inspector → "Graph Editor" tıkla
// 2. Sağ tık → Add Node → Code → Custom Code
// 3. Bu kodu Custom Code node'una yapıştır
// 4. Outputs'u Shader'ın Base Color ve Opacity'sine bağla
// ============================================================================

// INPUTS (Create these as Float/Color inputs in Graph)
// - progress: Float (0-1)
// - ringWidth: Float (default 0.2)
// - ringColor: Color
// - bgColor: Color

// CODE FOR CUSTOM CODE NODE:

vec2 uv = (fragCoord - 0.5) * 2.0;
float dist = length(uv);
float angle = atan(uv.x, uv.y);
float normAngle = fract((angle + 3.14159) / 6.28318 + 0.5);

float outer = 1.0;
float inner = 1.0 - ringWidth;
float smooth = 0.02;

float ring = smoothstep(inner - smooth, inner, dist) * (1.0 - smoothstep(outer, outer + smooth, dist));
float arc = 1.0 - smoothstep(progress - smooth, progress + smooth, normAngle);

vec3 color = mix(bgColor.rgb, ringColor.rgb, arc);
float alpha = ring * mix(bgColor.a, ringColor.a, arc);

// Output: vec4(color, alpha)
